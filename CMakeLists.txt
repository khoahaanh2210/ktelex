cmake_minimum_required(VERSION 3.16)
project(ktlex VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE core)
add_test(NAME telex COMMAND core_test)
add_executable(vowels_test tests/vowels_test.cpp)
target_include_directories(vowels_test PRIVATE core)
add_test(NAME vowels COMMAND vowels_test)
add_executable(late_modifiers_test tests/late_modifiers_test.cpp)
target_include_directories(late_modifiers_test PRIVATE core)
add_test(NAME late_modifiers COMMAND late_modifiers_test)
add_executable(syllables_test tests/syllables_test.cpp)
target_include_directories(syllables_test PRIVATE core)
add_test(NAME syllables COMMAND syllables_test)
add_executable(restore_test tests/restore_test.cpp)
target_include_directories(restore_test PRIVATE core)
add_test(NAME restore COMMAND restore_test)
add_test(NAME foreign_words COMMAND restore_test foreign)
add_test(NAME late_shape COMMAND restore_test late-shape)
add_test(NAME repeated_vowels COMMAND restore_test repeated-vowels)
option(BUILD_IBUS "Build IBus integration" ON)
if(BUILD_IBUS)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0)
 add_executable(ktlex ibus/main.cpp)
 target_include_directories(ktlex PRIVATE core)
 target_link_libraries(ktlex PRIVATE PkgConfig::IBUS)
 add_executable(ibus_test tests/ibus_test.cpp)
 target_include_directories(ibus_test PRIVATE core)
 target_link_libraries(ibus_test PRIVATE PkgConfig::IBUS)
 target_link_options(ibus_test PRIVATE -Wl,--wrap=ibus_engine_commit_text -Wl,--wrap=ibus_engine_update_preedit_text -Wl,--wrap=ibus_engine_update_preedit_text_with_mode)
 add_test(NAME ibus_space COMMAND ibus_test)
 add_executable(composition_test tests/composition_test.cpp)
 target_include_directories(composition_test PRIVATE core)
 target_link_libraries(composition_test PRIVATE PkgConfig::IBUS)
 target_link_options(composition_test PRIVATE -Wl,--wrap=ibus_engine_commit_text -Wl,--wrap=ibus_engine_update_preedit_text -Wl,--wrap=ibus_engine_delete_surrounding_text -Wl,--wrap=ibus_engine_update_preedit_text_with_mode)
 add_test(NAME ibus_composition COMMAND composition_test)
 option(BUILD_IBUS_DBUS_TESTS "Test through an isolated IBus daemon (requires Python GI)" OFF)
 if(BUILD_IBUS_DBUS_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(DBUS_RUN_SESSION dbus-run-session)
  find_program(IBUS_DAEMON ibus-daemon)
  if(NOT DBUS_RUN_SESSION OR NOT IBUS_DAEMON)
   message(FATAL_ERROR "D-Bus integration tests require dbus-run-session and ibus-daemon")
  endif()
  add_test(NAME ibus_dbus COMMAND ${DBUS_RUN_SESSION} -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ibus_dbus_test.py $<TARGET_FILE:ktlex> ${IBUS_DAEMON})
  set_tests_properties(ibus_dbus PROPERTIES TIMEOUT 45)
 endif()
 install(TARGETS ktlex DESTINATION libexec)
 configure_file(ibus/ktlex.xml.in ktlex.xml @ONLY)
 install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ktlex.xml DESTINATION share/ibus/component)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "ktlex maintainers")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libibus-1.0-5, libglib2.0-0, libstdc++6")
include(CPack)
